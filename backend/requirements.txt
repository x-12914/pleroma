# Pleroma backend requirements.
#
# Pin philosophy: only pin where there's a known cross-version constraint.
# The original repo pinned everything to versions authored for Python 3.11;
# on Python 3.14 those pins force source builds or hit assertion errors
# (e.g. SQLAlchemy 2.0.27 vs Python 3.14's new __firstlineno__ attr).
# Letting pip pick current versions gets us prebuilt wheels for 3.14.

# Web
fastapi
uvicorn[standard]
gunicorn

# Database
sqlalchemy
psycopg2-binary
pgvector

# Auth — using bcrypt directly (no passlib). passlib 1.7.4 is unmaintained
# and breaks against bcrypt 4.1+ due to a removed __about__ attribute.
python-jose[cryptography]
bcrypt

# Settings / validation
pydantic-settings
email-validator
python-multipart
python-dotenv

# HTTP + scraping
httpx
beautifulsoup4

# ML
joblib
scikit-learn
pandas
numpy

# Misc
slowapi
dateparser
